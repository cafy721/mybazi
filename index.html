<script>

    // 師門地支藏干表

    const customHiddenStems = {

        "子": ["癸", "辛", "壬"], "丑": ["己", "辛", "癸"], "寅": ["甲", "丙", "己"],

        "卯": ["乙", "癸", "甲"], "辰": ["己", "癸", "乙"], "巳": ["丙", "庚", "己"],

        "午": ["丁", "乙", "丙"], "未": ["戊", "乙", "丁"], "申": ["庚", "壬", "戊"],

        "酉": ["辛", "丁", "庚"], "戌": ["戊", "丁", "辛"], "亥": ["壬", "甲", "戊"]

    };


    const stemsData = {

        '甲': { element: '木', polarity: 1 }, '乙': { element: '木', polarity: 0 },

        '丙': { element: '火', polarity: 1 }, '丁': { element: '火', polarity: 0 },

        '戊': { element: '土', polarity: 1 }, '己': { element: '土', polarity: 0 },

        '庚': { element: '金', polarity: 1 }, '辛': { element: '金', polarity: 0 },

        '壬': { element: '水', polarity: 1 }, '癸': { element: '水', polarity: 0 }

    };


    const generates = { '木': '火', '火': '土', '土': '金', '金': '水', '水': '木' };

    const controls = { '木': '土', '土': '水', '水': '火', '火': '金', '金': '木' };


    function getTenGod(dayMaster, targetStem) {

        if (!dayMaster || !targetStem) return "";

        const dm = stemsData[dayMaster];

        const target = stemsData[targetStem];

        const isSamePolarity = dm.polarity === target.polarity;

        

        if (dm.element === target.element) return isSamePolarity ? '比肩' : '劫財';

        if (generates[dm.element] === target.element) return isSamePolarity ? '食神' : '傷官';

        if (generates[target.element] === dm.element) return isSamePolarity ? '偏印' : '正印';

        if (controls[dm.element] === target.element) return isSamePolarity ? '偏財' : '正財';

        if (controls[target.element] === dm.element) return isSamePolarity ? '七殺' : '正官';

        return "";

    }


    function renderHiddenStems(dayMaster, branch) {

        let hiddenStems = customHiddenStems[branch] || [];

        let html = '<div class="hidden-stems-container">';

        hiddenStems.forEach(hs => {

            html += `<div class="hidden-item"><span class="hidden-stem-char">${hs}</span><span class="ten-god">${getTenGod(dayMaster, hs)}</span></div>`;

        });

        html += '</div>';

        return html;

    }


    function calculateBazi() {

        const gender = parseInt(document.getElementById('gender').value);

        const y = parseInt(document.getElementById('year').value);

        const m = parseInt(document.getElementById('month').value);

        const d = parseInt(document.getElementById('day').value);

        const h = parseInt(document.getElementById('hour').value);

        const min = parseInt(document.getElementById('minute').value);


        const solar = Solar.fromYmdHms(y, m, d, h, min, 0);

        const lunar = solar.getLunar();

        const bazi = lunar.getEightChar();

        const yun = bazi.getYun(gender); // 根據性別起運


        const pillars = [

            { name: "時", stem: bazi.getTimeGan(), branch: bazi.getTimeZhi() },

            { name: "日", stem: bazi.getDayGan(), branch: bazi.getDayZhi() },

            { name: "月", stem: bazi.getMonthGan(), branch: bazi.getMonthZhi() },

            { name: "年", stem: bazi.getYearGan(), branch: bazi.getYearZhi() }

        ];


        const dayMaster = pillars[1].stem; 


        // 1. 渲染原局

        let htmlTopGods = '', htmlStems = '', htmlBranches = '', htmlHidden = '';

        pillars.forEach((p, index) => {

            htmlTopGods += `<td><span class="ten-god">${index === 1 ? '日主' : getTenGod(dayMaster, p.stem)}</span></td>`;

            htmlStems += `<td><div class="bazi-text">${p.stem}</div></td>`;

            htmlBranches += `<td><div class="bazi-text">${p.branch}</div></td>`;

            htmlHidden += `<td>${renderHiddenStems(dayMaster, p.branch)}</td>`;

        });

        document.getElementById('row-top-gods').innerHTML = htmlTopGods;

        document.getElementById('row-stems').innerHTML = htmlStems;

        document.getElementById('row-branches').innerHTML = htmlBranches;

        document.getElementById('row-hidden').innerHTML = htmlHidden;


        // 2. 渲染大運與流年 (取首 8 柱大運)

        const daYunArr = yun.getDaYun();

        let yunHtml = '<tr>';

        

        // 略過第一個預設小運 (index 0通常是起運前)，由 index 1 開始

        for (let i = 1; i <= 8 && i < daYunArr.length; i++) {

            const dy = daYunArr[i];

            const dyStem = dy.getGan();

            const dyBranch = dy.getZhi();

            

            // 獲取此大運的流年

            const liuNianArr = dy.getLiuNian();

            let lnStr = '<div class="liunian-list">';

            liuNianArr.forEach(ln => {

                lnStr += `<div class="liunian-item"><span>${ln.getYear()}年</span> <span>${ln.getGanZhi()}</span></div>`;

            });

            lnStr += '</div>';


            yunHtml += `

                <td>

                    <div class="yun-info">${dy.getStartAge()}歲 (${dy.getStartYear()}年)</div>

                    <span class="ten-god">${getTenGod(dayMaster, dyStem)}</span>

                    <div class="bazi-text">${dyStem}<br>${dyBranch}</div>

                    ${renderHiddenStems(dayMaster, dyBranch)}

                    ${lnStr}

                </td>

            `;

        }

        yunHtml += '</tr>';

        document.getElementById('dayun-table').innerHTML = yunHtml;


        document.getElementById('result').style.display = 'block';

    }

</script>
